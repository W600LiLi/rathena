	}

function	script	F_IsBlockedMap_ASIA	{
	.@map$ = getarg(0);

	for (.@i = 0; .@i < getarraysize(getvariableofnpc(.Blocked$, "AsiasoftEra_Blocker")); .@i++) {
		.@blk$ = getvariableofnpc(.Blocked$[.@i], "AsiasoftEra_Blocker");
		if (.@blk$ == "")
			continue;
		if (.@map$ == .@blk$)
			return 1;
	}
	return 0;
}
